cmake_minimum_required(VERSION 3.20)
project(vanius_windows_audio_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(audio_policy STATIC src/policy.cpp)
target_include_directories(audio_policy PUBLIC include)
add_executable(audio_policy_tests tests/policy_tests.cpp)
target_link_libraries(audio_policy_tests PRIVATE audio_policy)
enable_testing()
add_test(NAME policy_mocks_only COMMAND audio_policy_tests --receipt ${CMAKE_CURRENT_BINARY_DIR}/policy-mocks-receipt.json)
if(WIN32)
  add_library(windows_audio_adapter STATIC src/windows_adapter.cpp)
  target_include_directories(windows_audio_adapter PUBLIC include)
  target_compile_definitions(windows_audio_adapter PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(windows_audio_adapter PUBLIC audio_policy ole32 uuid)
  add_executable(windows_audio_candidate src/main.cpp)
  target_link_libraries(windows_audio_candidate PRIVATE windows_audio_adapter)
  # This test prints static capabilities; it does not instantiate COM/audio APIs.
  add_test(NAME capabilities_no_audio COMMAND windows_audio_candidate --capabilities)
endif()
if(MSVC)
  target_compile_options(audio_policy PRIVATE /W4 /WX /permissive-)
  target_compile_options(audio_policy_tests PRIVATE /W4 /WX /permissive-)
  if(WIN32)
    target_compile_options(windows_audio_adapter PRIVATE /W4 /WX /permissive-)
    target_compile_options(windows_audio_candidate PRIVATE /W4 /WX /permissive-)
  endif()
else()
  target_compile_options(audio_policy PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(audio_policy_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
